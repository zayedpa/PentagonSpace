------

'
'

CREATE TABLE STUDENT

BRANCH:

BID       INT         UNSIGNED NN PRIMARY KEY
BNAME     VARCHAR(10)          NN
IFSC_CODE VARCHAR(20)          NN UNIQUE,CHECK(LENGTH(IFSC_CODE)=11)
LOC       VARCHAR(10)          NN

PRIMARY KEY:P_BID 
UNIQUE:U_IFSC
CHECK:C_IFSC

CREATE TABLE BRANCH
(BID INT UNSIGNED NOT NULL,
 BNAME VARCHAR(10) NOT NULL,
 IFSC_CODE VARCHAR(20) NOT NULL,
 LOC VARCHAR(10) NOT NULL,
CONSTRAINT P_BID PRIMARY KEY(BID),
CONSTRAINT U_IFSC UNIQUE(IFSC_CODE),
CONSTRAINT C_IFSC CHECK(LENGTH(IFSC_CODE)=11)
);

PRODUCT:

PID      INT          NN  PRIMARY KEY
PNAME    VARCHAR(10)  NN
PRICE    DECIMAL(7,2) NN  CHECK(PRICE>0)
QUANTITY INT          NULL


PRIMARY KEY:P_PID
CHECK:C_PRICE

CREATE TABLE PRODUCT
(
PID INT NOT NULL,
PNAME VARCHAR(10) NOT NULL,
PRICE DECIMAL(7,2) NOT NULL,
QUANTITY INT,
CONSTRAINT P_PID PRIMARY KEY(PID),
CONSTRAINT C_PRICE CHECK(PRICE>0)
);

FACULTY:

FID    INT          UNSIGNED NN PRIMARY KEY 
FNAME  VARCHAR(10)           NN 
SALARY DECIMAL(5,2)          NN

CREATE TABLE FACULTY
(

SALARY DECIMAL(5,2) NOT NULL,
PHONE INT UNSIGNED UNIQUE,CHECK(LENGTH(PHONE)=10)

CUSTOMER:

CID      INT                   UNSIGNED  NN   PRIMARY KEY AUTO_INCREMENT
CNAME    VARCHAR(10)                     NN
PHONE    INT                   UNSIGNED  NN   UNIQUE,CHECK(LENGTH(PHONE)=10)
ADDRESS  VARCHAR(50)                     NN
GENDER   ENUM('MALE','FEMALE')           NN
BALANCE  DECIMAL(7,2)                    NULL DEFAULT'100'
ORDER_ID INT                   UNSIGNED  NULL 
ID       INT                   UNSIGNED  NULL  FOREIGN KEY

CREATE TABLE CUSTOMER
(
CID INT UNSIGNED  PRIMARY KEY AUTO_INCREMENT,
CNAME VARCHAR(10) NOT NULL,
PHONE INT UNSIGNED NOT NULL UNIQUE,CHECK(LENGTH(PHONE)=10),
ADDRESS VARCHAR(50) NOT NULL,
GENDER ENUM('MALE','FEMALE') NOT NULL,
BALANCE DECIMAL(7,2) NULL DEFAULT'100',
ORDER_ID INT UNSIGNED,
ID INT UNSIGNED,
CONSTRAINT FOREIGN KEY(ID)REFERENCES STUDENT(ID)
);

BANK:

ACCOUNT_NUMBER      INT          NN PRIMARY KEY
ACCOUNT_HOLDER_NAME VARCHAR(20)  NN 
BALANCE             DECIMAL(5,2) NN DEFAULT'500'
AREA                VARCHAR(10)  NN
BID             INT UNSIGNED    NULL    FOREIGN KEY

CREATE TABLE BANK
(
ACCOUNT_NUMBER INT PRIMARY KEY,
ACCOUNT_HOLDER_NAME VARCHAR(20) NOT NULL,
BALANCE DECIMAL(5,2) NOT NULL DEFAULT'500',
AREA VARCHAR(10) NOT NULL,
BID INT UNSIGNED,
CONSTRAINT FOREIGN KEY(BID)REFERENCES BRANCH(BID)
);

ALTER
-----
IT IS USED TO MODIFY STRUCTURE OF THE TABLE.

1.TO ADD NEW COLUMN
-------------------
SYNTAX:

ALTER TABLE TABLE_NAME
ADD NEW_COLUMN_NAME NEW_dATATYPE NULL/NOT NULL;

BALANCE DECIMAL(5,2) NULL----STUDENT

ALTER TABLE STUDENT
ADD BALANCE DECIMAL(5,2) NULL;

2.TO ADD NEW COLUMN AFTER PARTICULAR COLUMN
-------------------------------------------
SYNTAX:

ALTER TABLE TABLE_NAME
ADD NEW_COLUMN_NAME DATATYPE NULL/NOT NULL AFTER COLUMN_NAME;

GENDER VARCHAR(10) NULL---ID---STUDENT

ALTER TABLE STUDENT
ADD GENDER VARCHAR(10) NULL AFTER ID;

3.TO DROP COLUMN
----------------
SYNTAX:

ALTER TABLE TABLE_NAME
DROP COLUMN_NAME;

ALTER TABLE CUSTOMER
DROP ORDER_ID;

4.TO CHANGE DATATYPE
--------------------
SYNTAX:

ALTER TABLE TABLE_NAME
MODIFY EXISTING_COLUMN_NAME NEW_DATATYPE NULL/NOT NULL;

char(20)

alter table CUSTOMER
MODIFY CNAME CHAR(20) NOT NULL;

5.TO CHANGE NULL/NOT NULL
-------------------------
SYNTAX:

ALTER TABLE TABLE_NAME
MODIFY EXISTING_COLUMN_NAME EXISTING_DATATYPE NOT NULL/NULL;

ALTER TABLE CUSTOMER
MODIFY BALANCE DECIMAL(7,2) NOT NULL;

6.TO CHANGE TABLE_NAME
---------------------

SYNTAX:

ALTER TABLE TABLE_NAME
RENAME NEW_TABLE_NAME;

STUDENT:STUDENT_DATA

ALTER TABLE STUDENT
RENAME STUDENT_DATA;

7.TO CHANGE COLUMN_NAME
-----------------------
SYNTAX:

ALTER TABLE TABLE_NAME
CHANGE EXISTING_COLUMN_NAME NEW_COLUMN_NAME EXISTING_dATATYPE NULL/NOT NULL;

NAME:SNAME

ALTER TABLE STUDENT_DATA
CHANGE NAME SNAME VARCHAR(10) NOT NULL;

PHONE:CONTACT_NUMBER

ALTER TABLE STUDENT_DATA
CHANGE PHONE CONTACT_NUMBER BIGINT UNSIGNED NOT NULL;

ERROR:WE CANT DROP/RENAME A COLUMN IF COLUMN IS ASSIGNED WITH CHECK CONSTRAINT.

8.TO ADD CONSTRAINTS:
--------------------

i.TO ADD PRIMARY KEY
--------------------
SYNTAX:

ALTER TABLE TABLE_NAME
ADD PRIMARY KEY(COLUMN_NAME);

ii.TO ADD UNIQUE 
----------------
SYNTAX:

ALTER TABLE TABLE_NAME
ADD UNIQUE(COLUMN_NAME);

ALTER TABLE BANK
ADD UNIQUE(AREA);

iii.TO ADD CHECK
-----------------
SYNTAX:

ALTER TABLE TABLE_NAME
ADD CHECK(CONDITION);

IV.TO ADD FOREIGN KEY
---------------------
SYNTAX:

ALTER TABLE TABLE_NAME
ADD FOREIGN KEY(COLUMN_NAME)REFERENCES PARENT_TABLE_NAME(COLUMN_NAME);

ALTER TABLE FACULTY
ADD CID INT UNSIGNED NOT NULL;

ALTER TABLE FACULTY
ADD FOREIGN KEY(CID)REFERENCES CUSTOMER(CID);


V:TO ADD DEFAULT FOR EXISTING COLUMN
-------------------------------------
SYNTAX:

ALTER TABLE TABLE_NAME
MODIFY EXISTING_COLUMN_NAME EXISTING_DATATYPE NULL/NOT NULL DEFAULT'DEFAULT_VALUE';

ALTER TABLE FACULTY
MODIFY SALARY DECIMAL(5,2) NOT NULL DEFAULT'100';

ALTER TABLE PRODUCT
MODIFY QUANTITY INT NULL DEFAULT'10';

VI:TO ADD AUTO_INCREMENT FOR EXISTING COLUMN
--------------------------------------------
SYNTAX:

ALTER TABLE TABLE_NAME
MODIFY EXISTING_COLUMN_NAME EXISTING_dATATYPE NULL/NOT NULL AUTO_INCREMENT;

ALTER TABLE PRODUCT
MODIFY PID INT NOT NULL AUTO_INCREMENT;

TO DROP CONSTRAINTS:
------------------------

i. TO DROP PRIMARY KEY:
------------------------

ALTER TABLE TABLE_NAME
DROP PRIMARY KEY;

ALTER TABLE FACULTY
DROP PRIMARY KEY;

ALTER TABLE BRANCH
DROP PRIMARY KEY;

ERROR: WE CANT DROP PRIMARY KEY IF PRIMARY KEY COLUMN IS ACTING AS FOREIGN KEY IN ANOTHER TABLE

ALTER TABLE PRODUCT
DROP PRIMARY KEY;

ERROR: WE CANT DROP PRIMARY KEY FROM A TABLE IF PRIMARY KEY COLUMN IS ASSIGNED WITH AUTO_INCREMENT

ii. TO DROP DEFAULT:
---------------------
SYNTAX:

ALTER TABLE TABLE_NAME
ALTER COLUMN COLUMN_NAME
DROP DEFAULT;

ALTER TABLE BANK
ALTER COLUMN BALANCE
DROP DEFAULT;

iii. TO DROP AUTO-INCREMENT:
----------------------------
SYNTAX:

ALTER TABLE TABLE_NAME
MODIFY EXISTING_COLUMN_NAME EXISTING_DATATYPE NULL/NOT NULL;

ALTER TABLE PRODUCT
MODIFY PID INT NOT NULL;

iv. TO DROP OTHER CONSTRAINTS(UNIQUE, CHECK AND FOREIGN KEY):
-------------------------------------------------------------
SYNTAX:

ALTER TABLE TABLE_NAME
DROP CONSTRAINT CONSTRAINT_NAME;

ALTER TABLE BANK
DROP CONSTRAINT AREA;

ALIASING
---------
IT IS USED TO PROVIDE ALTERNATIVE NAME FOR A COLUMN IN RESULTANT TABLE.

RULES
-----
1.WITH OR WITHOUT USING AS KEYWORD WE CAN WRITE ALIAS NAME.
2.WE CAN USE MULTIPLE WORDS AS A ALIAS NAME BY CONNECTING IT WITH UNDER_SCORE OR BY   PASSING INSIDE THE QUOTES

SELECT SAL AS SALARY
FROM EMPS;                           :VALID

SELECT SAL SALARY
FROM EMPS;                           :VALID

SELECT SAL*12 AS 'ANNUAL SALARY'
FROM EMPS;                           :VALID

SELECT SAL*12 AS ANNUAL SALARY
FROM EMPS;                          :NOT VALID

SELECT SAL*12 ANNUAL_SALARY
FROM EMPS;                          :VALID

SELECT FNAME,JOB,COMM+(COMM*4/100) AS 'COMM WITH 4% HIKE',(SAL*12)-((SAL*12) * 8/100) 'ANNUAL SALARY WITH 8% DEDUCTION'

TYPES OF AGGREGATE FUNCTION
----------------------------
1.MAX():
--------

--------

SYNTAX: MIN(COLUMN_NAME/EXPRESSION)

FROM EMPS;      O/P:20000

3.AVG()
-------
IT IS USED TO OBTAIN AVERAGE VALUE FROM THE GIVEN COLUMN.

SYNTAX: AVG(COLUMN_NAME/EXPRESSION)

EX:

SELECT AVG(SAL)
FROM EMPS;     O/P:51000

--------

SYNTAX: SUM(COLUMN_NAME/EXPRESSION)

----------
IT IS USED TO OBTAIN NUMBER OF VALUES PRESENT IN GIVEN COLUMN.

SYNTAX: COUNT(*/COLUMN_NAME/EXPRESSION)

EX:

SELECT COUNT(SAL)
FROM EMPS;        O/P:10

FROM EMPS;     O/P:10

NOTE: ONLY FOR COUNT() WE CAN USE * AS A ARGUMENT.

CHARACTERISTICS OF AGGREGATE FUNCTION
-------------------------------------
>IT TAKES N NUMBER OF INPUTS BUT GENERATE SINGLE O/P.
>IT EXECUTES GROUP BY GROUP.
>WE CANT USE NORMAL COLUMNS ALONG WITH AGGREGATE FUNCTION IN SELECT CLAUSE.
>WE CAN PASS ONLY ONE COLUMN AS A ARGUMENT INSIDE AGGREGATE FUNCTION.
>WE CANT NEST AGGREGATE FUNCTIONS.
>WE CANT USE AGGREGATE FUNCTIONS INSIDE WHERE CLAUSE.
>IT IGNORE'S NULL VALUE.
>WE CAN USE GROUP BY EXPRESSION ALONG WITH AGGREGATE FUNCTION IN SELECT CLAUSE.

ORDER BY
--------
IT IS USED TO ARRENGE THE RECORDS EITHER IN ASCENDING OR DESCENDING ORDER.

SYNTAX:
-------
SELECT COLUMN_NAME/EXPRESSION
FROM TABLE_NAME
ORDER BY COLUMN_NAME ASC/DESC;

CHARACTERISTICS OF ORDER BY CLAUSE
----------------------------------
1.IT IS A LAST EXECUTABLE CLAUSE IN A QUERY.
2.IT EXECUTES AFTER SELECT CLAUSE.
3.BY DEFAULT IT WILL CONSIDER ASCENDING ORDER FOR COLUMNS.
4.NORMALLY IN ALL THE TABLES RECORDS ARE ARRENGED IN ASCENDING ORDER BASED ON 
  PRIMARY KEY COLUMN.
5.WE CAN USE ALIAS NAME INSIDE ORDER BY CLAUSE
6.WE CAN USE MULTIPLE COLUMNS INSIDE ORDER BY CLAUSE,IT WILL GIVE THE PRIORITY 
  FOR FIRST COLUMN ORDER,IF THE VALUES ARE SAME FOR FIRST COLUMN ORDER THEN IT 
  WILL GIVE THE PRIORITY FOR SECOND COLUMN ORDER.

ORDER OF EXECUTION
------------------
1.FROM
2.[WHERE]
3.GROUP BY
4.HAVING
5.SELECT 
6.ORDER BY

TRANSACTION CONTROL LANGUAGE
----------------------------

1.COMMIT
--------
TO SAVE ALL THE TRANSACTIONS PERMENENTLY INSIDE THE DATABASE WE USE COMMIT.

SYNTAX:

COMMIT;

2.ROLLBACK
-----------
TO ROLL OUT THE OPERATIONS UPTO PREVIOUSELY USED COMMIT STATEMENT WE USE ROLLBACK.

SYNTAX:
--------
ROLLBACK;

>WE CAN GETBACK THE DELETED RECORDS BY USING ROLLBACK,IF THE COMMIT IS NOT USED AFTER DELETE OPERATION

>WE CAN ROLL OUT THE OPERATIONS UPTO CERTAIN SAVEPOINTS.

SYNTAX:
-------
ROLLBACK TO SAVEPOINT_NAME;

SAVEPOINT

RANKING FUNCTIONS/WINDOW FUNCTIONS
-----------------------------------
IT IS USED ASSIGN RANKS FOR ALL THE RECORDS IN A TABLE.

SYNTAX:
------

SELECT RANKING_FUNCTION() OVER([PARTITION BY COLUMN_NAME] ORDER BY COLUMN_NAME ASC/DESC)
FROM TABLE_NAME;

OVER:IT IS USED TO PASS RANKING FUNCTIONS INSIDE SELECT CLAUSE.

PARTITION BY:IT IS USED TO CREATE GROUPS,IT RESETS THE RANK AFTER EACH GROUP.

TYPES OF RANKING FUNCTIOMS:
---------------------------
1.ROW_NUMBER()
2.RANK()
3.DENSE_RANK()

1.ROW_NUMBER()
--------------
IT IS USED TO ASSIGN UNIQUE RANKS FOR ALL THE RECORDS IN A TABLE.

SYNTAX:

SELECT ROW_NUMBER() OVER([PARTITION BY COLUMN_NAME] ORDER BY COLUMN_NAME ASC/DESC)
FROM TABLE_NAME;

SELECT FNAME,JOB,SAL,ROW_NUMBER() OVER(PARTITION BY JOB ORDER BY SAL DESC) AS 'RANK'
FROM EMPS;

SELECT FNAME,JOB,SAL,ROW_NUMBER() OVER(ORDER BY SAL DESC) AS 'RANK'
FROM EMPS;

drawback:
---------
IT WILL ASSIGN UNIQUE RANKS FOR TIED RECORDS.

2.RANK():
---------
IT IS USED TO ASSIGN RANKS FOR ALL THE RECORDS IN A TABLE.

>HERE,IT WILL ASSIGN SAME RANK FOR TIED RECORDS.

SYNTAX:

SELECT RANK() OVER([PARTITION BY COLUMN_NAME] ORDER BY COLUMN_NAME ASC/DESC)
FROM TABLE_NAME;

SELECT FNAME,JOB,SAL,RANK() OVER(ORDER BY SAL DESC) AS 'RANK'
FROM EMPS;

DRAWBACK:
---------
HERE,IT WILL ASSIGN SAME RANKS FOR TIED RECORDS BUT IT SKIPS NEXT RANKING NUMBERS.

3.DENSE_RANK():
---------------
IT IS USED TO ASSIGN RANKS FOR ALL THE RECORDS IN A TABLE.

>IT WILL ASSIGN SAME RANKS FOR TIED RECORDS ALSO IT REMAINS NEXT RANKING NUMBERS IN SEQUENTIAL ORDER.

SYNTAX:

SELECT DENSE_RANK() OVER([PARTITION BY COLUMN_NAME] ORDER BY COLUMN_NAME ASC/DESC)
FROM TABLE_NAME;

SELECT FNAME,JOB,SAL,DENSE_RANK() OVER(ORDER BY SAL DESC) AS 'RANK'
FROM EMPS;
